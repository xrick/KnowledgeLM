# 不使用 regex，用程式邏輯實現 pattern
class LogicBasedPattern:
    NUMBERS = ['二', '三', '四', '五', '六', '七', '八', '九', '十']
    UNITS = ['份', '篇', '個', '本']
    
    def detect_quantity_pattern(self, query: str) -> bool:
        # Pattern: 檢查是否有「數字」緊接著「量詞」
        for i, char in enumerate(query):
            if char in self.NUMBERS:
                # 檢查下一個字元是否為量詞
                if i + 1 < len(query) and query[i + 1] in self.UNITS:
                    return True
        return False

# 使用
detector = LogicBasedPattern()
detector.detect_quantity_pattern("五份文件")  # ✅ True
detector.detect_quantity_pattern("七本書")    # ✅ True

# 使用字串處理實現 pattern
class StringParsingPattern:
    def detect_quantity_pattern(self, query: str) -> bool:
        # 分詞
        import jieba
        words = jieba.lcut(query)
        
        # Pattern: 檢查是否有數量詞
        for word in words:
            # 檢查詞性是否為數量詞
            if self.is_quantity_word(word):
                return True
        return False
    
    def is_quantity_word(self, word: str) -> bool:
        # 簡化版：包含數字+量詞
        number_chars = set('一二三四五六七八九十百千萬')
        unit_chars = set('份篇個本張支條')
        
        has_number = any(c in number_chars for c in word)
        has_unit = any(c in unit_chars for c in word)
        
        return has_number and has_unit
方法 C: NLP-based Pattern（語言學規則）
# 使用 NLP 工具實現 pattern
import spacy

class NLPPattern:
    def __init__(self):
        self.nlp = spacy.load("zh_core_web_sm")
    
    def detect_quantity_pattern(self, query: str) -> bool:
        doc = self.nlp(query)
        
        # Pattern: 檢查是否有 NUM (數字) 詞性
        for token in doc:
            if token.pos_ == "NUM":
                # 檢查下一個詞是否為量詞
                if token.i + 1 < len(doc):
                    next_token = doc[token.i + 1]
                    if next_token.pos_ == "M":  # M = 量詞 (Measure word)
                        return True
        return False
