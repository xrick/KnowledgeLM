CREATE TABLE file_metadata (
                    file_id TEXT PRIMARY KEY,
                    filename TEXT NOT NULL,
                    file_type TEXT NOT NULL,
                    file_size INTEGER,
                    upload_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                    user_id TEXT,
                    chunk_count INTEGER,
                    embedding_status TEXT,
                    milvus_partition TEXT,
                    metadata_json TEXT
                )
