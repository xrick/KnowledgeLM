實作「停止生成」功能 (Frontend)
我們需要利用 JavaScript 的 AbortController 來中斷 HTTP 請求，並在 UI 上將「搜尋」按鈕動態切換為「停止」按鈕。

1. 修改 HTML (按鈕部分)
找到您的 send-button (約 Line 780)，我們需要為它準備兩個圖示（搜尋與停止）。

    <i class="fa-solid fa-paper-plane" id="btn-icon-send"></i>
    <i class="fa-solid fa-stop" id="btn-icon-stop" style="display: none;"></i>
    <span id="btn-text">搜尋</span>
</button>
2. 修改 JavaScript (邏輯部分)
我們需要一個全域變數來儲存目前的 controller，並修改傳送邏輯。

請在 script 區塊的最上方（State 定義區）加入：

JavaScript

let currentAbortController = null; // 用來控制中斷
接著，新增/修改以下函式：

JavaScript

// 1. 統一的點擊處理函式
function handleSendClick() {
    if (isLoading) {
        // 如果正在載入，這個點擊代表「停止」
        stopGeneration();
    } else {
        // 如果閒置，這個點擊代表「送出」
        sendQuery();
    }
}

// 2. 停止生成的邏輯
function stopGeneration() {
    if (currentAbortController) {
        currentAbortController.abort(); // 發送中斷訊號
        currentAbortController = null;
        
        // 強制結束 UI 狀態
        endLoadingState();
        
        // 在對話框顯示「已停止」提示
        addMessage('⚠️ 使用者已中斷生成', 'system'); // 您可能需要 CSS 支援 system 訊息樣式
    }
}

// 3. 封裝 UI 狀態切換 (開始/結束)
function startLoadingState() {
    isLoading = true;
    const btnText = document.getElementById('btn-text');
    const iconSend = document.getElementById('btn-icon-send');
    const iconStop = document.getElementById('btn-icon-stop');
    const sendButton = document.getElementById('send-button');
    const input = document.getElementById('query-input');

    // 切換為停止樣式
    if(btnText) btnText.textContent = '停止';
    if(iconSend) iconSend.style.display = 'none';
    if(iconStop) iconStop.style.display = 'inline-block';
    
    // 注意：不要 disable 按鈕，否則無法點擊停止
    // sendButton.disabled = true;  <-- 移除這行
    sendButton.classList.add('loading-state'); // 可加 CSS 做紅底效果
    
    // 鎖定輸入框
    // input.disabled = true; // 可選，通常串流時不讓輸入
}

function endLoadingState() {
    isLoading = false;
    const btnText = document.getElementById('btn-text');
    const iconSend = document.getElementById('btn-icon-send');
    const iconStop = document.getElementById('btn-icon-stop');
    const sendButton = document.getElementById('send-button');
    const input = document.getElementById('query-input');

    // 切換回搜尋樣式
    if(btnText) btnText.textContent = '搜尋';
    if(iconSend) iconSend.style.display = 'inline-block';
    if(iconStop) iconStop.style.display = 'none';
    
    sendButton.classList.remove('loading-state');
    
    input.disabled = false;
    input.focus();
}
最後，修改您的 sendProgressiveQuery (或 sendTraditionalQuery) 來使用 AbortController：

JavaScript

async function sendProgressiveQuery(query) {
    // 初始化狀態
    startLoadingState();
    
    // 建立新的 Controller
    currentAbortController = new AbortController();
    const signal = currentAbortController.signal;

    try {
        // ... (中間的 UI 建立程式碼保持不變) ...

        // 在呼叫 fetch 或您的串流函式時，傳入 signal
        // 假設 startProgressiveChat 支援 fetch options
        await startProgressiveChat(
            query,
            url,
            containerSelector,
            progressSelector,
            docIds,
            (success) => { endLoadingState(); }, // 完成後的 callback
            signal // <--- 傳入 signal
        );

    } catch (error) {
        if (error.name === 'AbortError') {
            console.log('Fetch aborted by user');
            // 已在 stopGeneration 處理 UI，這裡不用做太多事
        } else {
            console.error('Error:', error);
            endLoadingState();
        }
    }
}
注意： 您需要在您的 startProgressiveChat 內部實作中，將這個 signal 傳遞給 fetch(url, { signal: signal, ... })，這樣瀏覽器才會真正切斷連線。
